name: Build & publish 11176.com
# Generates ~1,700 SEO number pages + sitemap + OG image, then publishes to the gh-pages branch.
# GitHub Pages: Settings → Pages → Source "Deploy from a branch" → gh-pages / (root).
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: write
concurrency:
  group: pages
  cancel-in-progress: true
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install pillow
      - run: python tools/generate.py
      - name: Publish _site to gh-pages
        run: |
          cd _site
          git init -q -b gh-pages
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add -A
          git commit -qm "Publish site from ${GITHUB_SHA::7}"
          git push -f "https://x-access-token:${{ secrets.GITHUB_TOKEN }}@github.com/${{ github.repository }}.git" gh-pages
